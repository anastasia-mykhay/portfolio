export const toRgb = (value) => {
  const ctx = document.createElement("canvas").getContext("2d");
  ctx.fillStyle = value;
  ctx.fillRect(0, 0, 1, 1);
  return Array.from(ctx.getImageData(0, 0, 1, 1).data.slice(0, 3));
};

export const luminance = ([r, g, b]) => {
  const [lr, lg, lb] = [r, g, b].map((c) => {
    const v = c / 255;
    return v <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4;
  });
  return 0.2126 * lr + 0.7152 * lg + 0.0722 * lb;
};

export const verdict = (ratio) =>
  ratio >= 7
    ? "AAA: підходить для будь-якого тексту"
    : ratio >= 4.5
      ? "AA: підходить для будь-якого тексту"
      : ratio >= 3
        ? "Лише великий текст (від 24 px або 19 px жирного)"
        : "Не проходить";

export const FontSize = ({ token }) => {
  const [text, setText] = useState("");
  useEffect(() => {
    const raw = getComputedStyle(document.documentElement)
      .getPropertyValue("--text-" + token)
      .trim();
    const m = raw.match(/^clamp\(\s*([\d.]+)px\s*,.+,\s*([\d.]+)px\s*\)$/);
    setText(m ? m[2] + " px" : raw.replace("px", " px"));
  }, [token]);
  return <span className="text-text-muted">{text}</span>;
};

export const ContrastTable = ({ pairs, mode = "text" }) => {
  const [rows, setRows] = useState([]);
  useEffect(() => {
    const read = (name) => {
      const probe = document.createElement("span");
      probe.style.color = "var(--color-" + name + ")";
      document.body.appendChild(probe);
      const resolved = getComputedStyle(probe).color;
      probe.remove();
      return toRgb(resolved);
    };
    setRows(
      pairs.map(([fg, bg, note]) => {
        const a = luminance(read(fg));
        const b = luminance(read(bg));
        const ratio = (Math.max(a, b) + 0.05) / (Math.min(a, b) + 0.05);
        return { fg, bg, ratio, note };
      }),
    );
  }, [pairs]);
  return (
    <Unstyled>
      <table className="w-full border-collapse text-small text-text">
        <thead>
          <tr className="border-b border-line text-left">
            <th className="py-2 pr-4">Пара (текст на фоні)</th>
            <th className="py-2 pr-4">Контраст</th>
            <th className="py-2">Висновок</th>
          </tr>
        </thead>
        <tbody>
          {rows.map(({ fg, bg, ratio, note }) => (
            <tr key={fg + bg} className="border-b border-line">
              <td className="py-2 pr-4">
                <span className="mr-2 inline-block rounded-control border border-line px-3 py-1" style={{ background: "var(--color-" + bg + ")", color: "var(--color-" + fg + ")" }}>Aa</span>
                {fg} на {bg}
              </td>
              <td className="py-2 pr-4">{ratio.toFixed(2)} : 1</td>
              <td className="py-2">
                {mode === "decorative"
                  ? "Не текст (" + note + "): за контрастом не оцінюється"
                  : mode === "forbidden"
                    ? "Не використовувати для тексту"
                    : verdict(ratio)}
              </td>
            </tr>
          ))}
        </tbody>
      </table>
    </Unstyled>
  );
};

## Контраст пар

Контраст рахується за формулою WCAG зі змінних `@theme`, значень у коді немає. Для звичайного тексту потрібно не менше 4.5 : 1, для великого не менше 3 : 1.

### Текст

<ContrastTable pairs={textPairs} />

### Не текст

Пари нижче мають контраст менше 4.5 : 1, але самі нічого не повідомляють: це декоративні елементи, тому вони не вада.

<ContrastTable pairs={decorativePairs} mode="decorative" />

### Не використовувати для тексту

Кольори проєктів лише для заливок і маркерів. Ці пари не проходять як текст.

<ContrastTable pairs={forbiddenPairs} mode="forbidden" />

## Шкала шрифтів

Шрифт: `font-sans` (Archivo, запасний Arial). Великі літери для `label` і `label-strong` додає компонент (`uppercase`).

Стилі `display-*` і `email` мають діапазон: розмір росте разом із шириною екрана, від найменшого до найбільшого. Адаптивні класи для цього не потрібні. Стилі `body`, `label`, `label-strong` і `small` мають один розмір на всіх екранах.

## Пропорції

Токенів `--aspect-*` у системі немає: скріншоти показуємо в природних пропорціях без обрізання.

## Рух кільця

Токени `spacing-ring-travel` (шлях кільця) і `ease-ring` (крива руху) призначені лише для руху кільця. Шлях кільця дорівнює трьом крокам «крапка + проміжок `gap-3`» і рахується від `spacing-dot`. Наведіть мишу на темний блок: кільце пройде свій шлях.

  <span className="text-small text-text-on-dark">translate-y-ring-travel, ease-ring</span>

</Unstyled>
